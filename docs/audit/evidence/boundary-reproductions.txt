import { readFileSync } from 'node:fs'
import path from 'node:path'
import { afterAll, describe, expect, it } from 'vitest'
import { BASE, PASSWORD, closeDb, createUser, db, login, uniqueIp } from '../support/helpers'

// Audit-only tests in the disposable copy; all records and mail are synthetic.
const manifest = JSON.parse(readFileSync(path.resolve(import.meta.dirname, '../../.next/server/server-reference-manifest.json'), 'utf8'))
function actionId(name: string): string {
  const found = Object.entries(manifest.node as Record<string, { exportedName?: string }>).find(([, value]) => value.exportedName === name)
  if (!found) throw new Error('Action absent from fresh manifest: ' + name)
  return found[0]
}
async function invoke(name: string, args: unknown[], cookie?: string) {
  const endpoint = name === 'listNotificationsForUser' ? '/contact' : name === 'listApplicationsForUser' ? '/dashboard/applications' : name === 'getApplicationForReview' ? '/admin/applications/audit-probe' : '/login'
  const response = await fetch(BASE() + endpoint, {
    method: 'POST',
    headers: { 'Next-Action': actionId(name), Accept: 'text/x-component', 'Content-Type': 'text/plain;charset=UTF-8', 'x-forwarded-for': uniqueIp(), ...(cookie ? { cookie } : {}) },
    body: JSON.stringify(args),
  })
  const body = await response.clone().text()
  console.info('AUDIT-TRANSPORT', JSON.stringify({name,status:response.status,contentType:response.headers.get('content-type'),length:body.length,privateNotice:body.includes('AUDIT-SYNTHETIC-PRIVATE-NOTICE'),privateNote:body.includes('AUDIT-SYNTHETIC-PRIVATE-DECISION'),passwordHashField:body.includes('"passwordHash":'),draft:body.includes('"status":"draft"'),emptyJson:body==='{}'}))
  return response
}
afterAll(closeDb)
describe('audit synthetic reproductions through actual Server Actions', () => {
  it('anonymous action reads another member notification', async () => {
    const id = await createUser({ email: 'audit-notification@example.test' })
    const marker = 'AUDIT-SYNTHETIC-PRIVATE-NOTICE'
    await db().query("insert into app.notifications (id,user_id,type,title,body) values(gen_random_uuid()::text,$1,'application_received',$2,'Synthetic only')", [id, marker])
    const response = await invoke('listNotificationsForUser', [id])
    const body = await response.text()
    expect(response.status).toBe(200)
    expect(body.includes(marker)).toBe(true)
  })
  it('anonymous action reads another member application and staff-only note', async () => {
    const id = await createUser({ email: 'audit-application@example.test' })
    const marker = 'AUDIT-SYNTHETIC-PRIVATE-DECISION'
    await db().query("insert into app.applications(id,user_id,program_id,decision_note) values(gen_random_uuid()::text,$1,2000000001,$2)", [id, marker])
    const response = await invoke('listApplicationsForUser', [id])
    const body = await response.text()
    expect(response.status).toBe(200)
    expect(body.includes(marker)).toBe(true)
  })
  it('route-only login registration does not demonstrate page-action CAPTCHA bypass', async () => {
    const email = 'audit-login@example.test'
    await createUser({ email })
    const route = await fetch(BASE() + '/api/auth/password/login', { method: 'POST', headers: { 'content-type': 'application/json', 'x-forwarded-for': uniqueIp() }, body: JSON.stringify({ email, password: PASSWORD }) })
    expect(route.status).toBeGreaterThanOrEqual(400)
    const response = await invoke('loginWithPassword', [{ email, password: PASSWORD }])
    const body = await response.text()
    expect(response.status).toBe(200)
    expect(body).toBe('{}')
    expect(response.headers.getSetCookie().some(c => c.includes('session-token='))).toBe(false)
  })
  it('reviewer action serializes a synthetic password hash and can read a draft', async () => {
    const applicantId = await createUser({ email: 'audit-review-applicant@example.test' })
    const staffEmail = 'audit-reviewer@example.test'
    await createUser({ email: staffEmail, staffRole: 'reviewer' })
    const session = await login(staffEmail)
    expect(Boolean(session)).toBe(true)
    const result = await db().query("insert into app.applications(id,user_id,program_id) values(gen_random_uuid()::text,$1,2000000002) returning id", [applicantId])
    const response = await invoke('getApplicationForReview', [result.rows[0].id], session!.cookie)
    const body = await response.text()
    expect(response.status).toBe(200)
    expect(body.includes('"passwordHash":')).toBe(true)
    expect(body.includes('"status":"draft"')).toBe(true)
  })
})
